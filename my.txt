# Installs the image's software machine-wide with winget. Runs as SYSTEM.
#
# winget ships with Windows 11 (the App Installer package) but cannot be
# registered for SYSTEM, so winget.exe is run straight from its install
# folder. Run that way it cannot load the VC++ runtime it normally gets from
# its package dependencies, so the VC++ redistributable is installed first.
$ErrorActionPreference = 'Stop'
$ProgressPreference    = 'SilentlyContinue'

$vcRedist = "$env:TEMP\vc_redist.x64.exe"
Invoke-WebRequest -Uri 'https://aka.ms/vs/17/release/vc_redist.x64.exe' -OutFile $vcRedist -UseBasicParsing
$exitCode = (Start-Process -FilePath $vcRedist -ArgumentList '/install /quiet /norestart' -Wait -PassThru).ExitCode
# 1638 = already installed, 3010 = reboot required (the build restarts after this script)
if ($exitCode -notin 0, 1638, 3010) { throw "VC++ redistributable install failed with exit code $exitCode" }
Remove-Item $vcRedist

$appInstaller = Get-AppxPackage -AllUsers -Name Microsoft.DesktopAppInstaller |
    Sort-Object { [version]$_.Version } | Select-Object -Last 1
if (-not $appInstaller) { throw 'winget (Microsoft.DesktopAppInstaller) is not on the base image' }
$winget = Join-Path $appInstaller.InstallLocation 'winget.exe'

foreach ($id in 'Google.Chrome', 'Mozilla.Firefox', 'Microsoft.AzureCLI', 'Microsoft.DotNet.SDK.10') {
    & $winget install --id $id --exact --source winget --scope machine --silent `
        --disable-interactivity --accept-package-agreements --accept-source-agreements
    if ($LASTEXITCODE -ne 0) { throw ('winget install {0} failed with exit code 0x{1:X8}' -f $id, $LASTEXITCODE) }
}
